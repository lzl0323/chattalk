## 🔧 技术实现

### 头像颜色生成算法
```javascript
const avatarColor = computed(() => {
  if (!currentUser.value?.email) return '#6366f1'
  
  // 预设的 12 种颜色
  const colors = [
    '#ef4444', '#f59e0b', '#10b981', '#3b82f6', 
    '#6366f1', '#8b5cf6', '#ec4899', '#14b8a6',
    '#f97316', '#84cc16', '#06b6d4', '#6366f1'
  ]
  
  // 使用邮箱生成哈希值
  let hash = 0
  for (let i = 0; i < currentUser.value.email.length; i++) {
    hash = currentUser.value.email.charCodeAt(i) + ((hash << 5) - hash)
  }
  
  return colors[Math.abs(hash) % colors.length]
})
```

### 标题自动生成
```javascript
async function autoGenerateTitle(messages) {
  // 获取第一条用户消息
  const firstUserMessage = messages.find(m => m.role === 'user')
  if (!firstUserMessage) return
  
  // 提取前 20 个字符作为标题
  const title = firstUserMessage.content.substring(0, 20) + 
                (firstUserMessage.content.length > 20 ? '...' : '')
  
  // 更新对话标题
  await updateConversation(currentConversationId.value, title)
}
```
